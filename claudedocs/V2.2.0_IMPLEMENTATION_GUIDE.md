```python
"""Authentication data models"""

from pydantic import BaseModel, EmailStr, Field, validator
from typing import Optional, List
from datetime import datetime
import re


class UserCreate(BaseModel):
    """회원가입 요청"""
    email: EmailStr
    password: str = Field(..., min_length=8)
    username: str = Field(..., min_length=2, max_length=50)

    @validator('password')
    def validate_password_strength(cls, v):
        """비밀번호 강도 검증"""
        if len(v) < 8:
            raise ValueError('비밀번호는 최소 8자 이상이어야 합니다')

        # 대문자, 소문자, 숫자, 특수문자 각 1개 이상
        if not re.search(r'[A-Z]', v):
            raise ValueError('비밀번호에 대문자가 포함되어야 합니다')
        if not re.search(r'[a-z]', v):
            raise ValueError('비밀번호에 소문자가 포함되어야 합니다')
        if not re.search(r'\d', v):
            raise ValueError('비밀번호에 숫자가 포함되어야 합니다')
        if not re.search(r'[!@#$%^&*(),.?":{}|<>]', v):
            raise ValueError('비밀번호에 특수문자가 포함되어야 합니다')

        return v

    @validator('username')
    def validate_username(cls, v):
        """사용자명 검증"""
        if not re.match(r'^[a-zA-Z0-9가-힣_]+$', v):
            raise ValueError('사용자명은 영문, 숫자, 한글, 언더스코어만 가능합니다')
        return v


class UserLogin(BaseModel):
    """로그인 요청"""
    email: EmailStr
    password: str


class User(BaseModel):
    """사용자 모델"""
    user_id: str
    email: EmailStr
    username: str
    created_at: datetime
    last_login: Optional[datetime] = None
    is_active: bool = True
    role: str = "user"  # user, admin
    failed_login_attempts: int = 0
    locked_until: Optional[datetime] = None


class TokenPair(BaseModel):
    """토큰 쌍"""
    access_token: str
    refresh_token: str
    token_type: str = "bearer"


class LoginResponse(BaseModel):
    """로그인 응답"""
    user: User
    tokens: TokenPair


class Session(BaseModel):
    """세션 모델"""
    session_id: str
    user_id: str
    created_at: datetime
    expires_at: datetime
    ip_address: Optional[str] = None
    user_agent: Optional[str] = None


class PasswordReset(BaseModel):
    """비밀번호 재설정 요청"""
    email: EmailStr


class PasswordResetConfirm(BaseModel):
    """비밀번호 재설정 확인"""
    token: str
    new_password: str = Field(..., min_length=8)
```

```python
"""Authentication utilities"""

from passlib.context import CryptContext
from jose import JWTError, jwt
from datetime import datetime, timedelta
from typing import Optional, Dict
import os

# 비밀번호 해싱
pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")

# JWT 설정
SECRET_KEY = os.getenv("JWT_SECRET_KEY", "your-secret-key-change-in-production")
ALGORITHM = os.getenv("JWT_ALGORITHM", "HS256")
ACCESS_TOKEN_EXPIRE_MINUTES = int(os.getenv("JWT_ACCESS_TOKEN_EXPIRE_MINUTES", "60"))
REFRESH_TOKEN_EXPIRE_DAYS = int(os.getenv("JWT_REFRESH_TOKEN_EXPIRE_DAYS", "7"))


def hash_password(password: str) -> str:
    """비밀번호 해싱"""
    return pwd_context.hash(password)


def verify_password(plain_password: str, hashed_password: str) -> bool:
    """비밀번호 검증"""
    return pwd_context.verify(plain_password, hashed_password)


def create_access_token(data: Dict, expires_delta: Optional[timedelta] = None) -> str:
    """Access Token 생성"""
    to_encode = data.copy()

    if expires_delta:
        expire = datetime.utcnow() + expires_delta
    else:
        expire = datetime.utcnow() + timedelta(minutes=ACCESS_TOKEN_EXPIRE_MINUTES)

    to_encode.update({"exp": expire, "type": "access"})
    encoded_jwt = jwt.encode(to_encode, SECRET_KEY, algorithm=ALGORITHM)

    return encoded_jwt


def create_refresh_token(data: Dict) -> str:
    """Refresh Token 생성"""
    to_encode = data.copy()
    expire = datetime.utcnow() + timedelta(days=REFRESH_TOKEN_EXPIRE_DAYS)

    to_encode.update({"exp": expire, "type": "refresh"})
    encoded_jwt = jwt.encode(to_encode, SECRET_KEY, algorithm=ALGORITHM)

    return encoded_jwt


def verify_token(token: str, expected_type: str = "access") -> Optional[Dict]:
    """토큰 검증 및 디코딩"""
    try:
        payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])

        # 토큰 타입 확인
        if payload.get("type") != expected_type:
            return None

        return payload

    except JWTError:
        return None


def create_token_pair(user_id: str) -> Dict[str, str]:
    """Access Token + Refresh Token 쌍 생성"""
    access_token = create_access_token(data={"user_id": user_id})
    refresh_token = create_refresh_token(data={"user_id": user_id})

    return {
        "access_token": access_token,
        "refresh_token": refresh_token,
        "token_type": "bearer"
    }
```

**`src/auth/service.py`**
```python
"""Authentication service layer"""

from typing import Optional, Dict
from datetime import datetime, timedelta
import uuid
import json
from .models import User, UserCreate, UserLogin, Session
from .utils import hash_password, verify_password, create_token_pair
from ..cache_manager import CacheManager


class AuthService:
    """사용자 인증 서비스"""

    def __init__(self, cache_manager: CacheManager):
        self.cache = cache_manager
        self.redis = cache_manager.redis_client

    async def create_user(self, user_data: UserCreate) -> User:
        """새 사용자 생성"""

        # 1. 이메일 중복 확인
        existing_user_id = await self.redis.get(f"user:email:{user_data.email}")
        if existing_user_id:
            raise ValueError("이미 사용 중인 이메일입니다")

        # 2. 사용자 ID 생성
        user_id = str(uuid.uuid4())

        # 3. 비밀번호 해싱
        password_hash = hash_password(user_data.password)

        # 4. 사용자 데이터 생성
        user = User(
            user_id=user_id,
            email=user_data.email,
            username=user_data.username,
            created_at=datetime.utcnow(),
            is_active=True,
            role="user"
        )

        # 5. Redis에 저장
        user_key = f"user:{user_id}"
        user_dict = user.dict()
        user_dict["password_hash"] = password_hash
        user_dict["created_at"] = user_dict["created_at"].isoformat()

        await self.redis.hset(user_key, mapping=user_dict)

        # 6. 이메일 인덱스 생성
        await self.redis.set(f"user:email:{user_data.email}", user_id)

        return user

    async def authenticate_user(
        self,
        credentials: UserLogin,
        ip_address: Optional[str] = None
    ) -> Dict:
        """사용자 인증"""

        # 1. 이메일로 사용자 조회
        user_id = await self.redis.get(f"user:email:{credentials.email}")
        if not user_id:
            raise ValueError("이메일 또는 비밀번호가 올바르지 않습니다")

        user_data = await self.redis.hgetall(f"user:{user_id}")

        # 2. 계정 잠금 확인
        if user_data.get("locked_until"):
            locked_until = datetime.fromisoformat(user_data["locked_until"])
            if datetime.utcnow() < locked_until:
                remaining = (locked_until - datetime.utcnow()).seconds // 60
                raise ValueError(f"계정이 잠겼습니다. {remaining}분 후 다시 시도하세요")
            else:
                # 잠금 해제
                await self.redis.hset(f"user:{user_id}", "locked_until", "")
                await self.redis.hset(f"user:{user_id}", "failed_login_attempts", "0")

        # 3. 비밀번호 검증
        if not verify_password(credentials.password, user_data["password_hash"]):
            # 실패 횟수 증가
            failed_attempts = int(user_data.get("failed_login_attempts", 0)) + 1
            await self.redis.hset(f"user:{user_id}", "failed_login_attempts", failed_attempts)

            # 5회 이상 실패 시 계정 잠금
            if failed_attempts >= 5:
                locked_until = datetime.utcnow() + timedelta(minutes=15)
                await self.redis.hset(
                    f"user:{user_id}",
                    "locked_until",
                    locked_until.isoformat()
                )
                raise ValueError("로그인 5회 실패로 계정이 15분간 잠겼습니다")

            raise ValueError("이메일 또는 비밀번호가 올바르지 않습니다")

        # 4. 활성 계정 확인
        if user_data.get("is_active") != "True":
            raise ValueError("비활성화된 계정입니다")

        # 5. 로그인 성공 처리
        await self.redis.hset(f"user:{user_id}", "last_login", datetime.utcnow().isoformat())
        await self.redis.hset(f"user:{user_id}", "failed_login_attempts", "0")
        await self.redis.hset(f"user:{user_id}", "locked_until", "")

        # 6. 토큰 생성
        tokens = create_token_pair(user_id)

        # 7. 세션 생성
        session = await self.create_session(
            user_id=user_id,
            access_token=tokens["access_token"],
            ip_address=ip_address
        )

        # 8. 사용자 객체 생성 (비밀번호 제외)
        user = User(
            user_id=user_data["user_id"],
            email=user_data["email"],
            username=user_data["username"],
            created_at=datetime.fromisoformat(user_data["created_at"]),
            last_login=datetime.utcnow(),
            is_active=user_data["is_active"] == "True",
            role=user_data.get("role", "user")
        )

        return {
            "user": user,
            "tokens": tokens,
            "session": session
        }

    async def create_session(
        self,
        user_id: str,
        access_token: str,
        ip_address: Optional[str] = None,
        user_agent: Optional[str] = None
    ) -> Session:
        """세션 생성"""

        session_id = str(uuid.uuid4())
        now = datetime.utcnow()
        expires_at = now + timedelta(hours=1)

        session = Session(
            session_id=session_id,
            user_id=user_id,
            created_at=now,
            expires_at=expires_at,
            ip_address=ip_address,
            user_agent=user_agent
        )

        # Redis에 저장 (TTL: 1시간)
        session_key = f"session:{session_id}"
        session_dict = session.dict()
        session_dict["created_at"] = session_dict["created_at"].isoformat()
        session_dict["expires_at"] = session_dict["expires_at"].isoformat()

        await self.redis.hset(session_key, mapping=session_dict)
        await self.redis.expire(session_key, 3600)  # 1시간

        # 사용자의 세션 목록에 추가
        await self.redis.sadd(f"user:sessions:{user_id}", session_id)

        return session

    async def logout(self, session_id: str) -> bool:
        """로그아웃"""

        # 세션 조회
        session_data = await self.redis.hgetall(f"session:{session_id}")
        if not session_data:
            return False

        user_id = session_data["user_id"]

        # 세션 삭제
        await self.redis.delete(f"session:{session_id}")
        await self.redis.srem(f"user:sessions:{user_id}", session_id)

        return True

    async def get_user_by_id(self, user_id: str) -> Optional[User]:
        """사용자 조회"""

        user_data = await self.redis.hgetall(f"user:{user_id}")
        if not user_data:
            return None

        return User(
            user_id=user_data["user_id"],
            email=user_data["email"],
            username=user_data["username"],
            created_at=datetime.fromisoformat(user_data["created_at"]),
            last_login=datetime.fromisoformat(user_data["last_login"]) if user_data.get("last_login") else None,
            is_active=user_data["is_active"] == "True",
            role=user_data.get("role", "user")
        )
```

**`src/auth/middleware.py`**
```python
"""Authentication middleware"""

from fastapi import Depends, HTTPException, status, Request
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials
from typing import Optional
from .utils import verify_token
from .service import AuthService
from ..cache_manager import CacheManager

# HTTP Bearer 토큰 스키마
security = HTTPBearer(auto_error=False)


async def get_current_user(
    request: Request,
    credentials: Optional[HTTPAuthorizationCredentials] = Depends(security)
) -> dict:
    """현재 사용자 조회 (JWT 검증)"""

    if not credentials:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="인증이 필요합니다",
            headers={"WWW-Authenticate": "Bearer"},
        )

    # JWT 토큰 검증
    token = credentials.credentials
    payload = verify_token(token, expected_type="access")

    if not payload:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="유효하지 않은 토큰입니다",
            headers={"WWW-Authenticate": "Bearer"},
        )

    # 사용자 조회
    cache_manager = request.app.state.cache_manager
    auth_service = AuthService(cache_manager)

    user_id = payload.get("user_id")
    user = await auth_service.get_user_by_id(user_id)

    if not user:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="사용자를 찾을 수 없습니다"
        )

    return user.dict()


async def get_current_active_user(
    current_user: dict = Depends(get_current_user)
) -> dict:
    """활성 사용자만 허용"""

    if not current_user.get("is_active"):
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="비활성화된 계정입니다"
        )

    return current_user


async def require_admin(
    current_user: dict = Depends(get_current_active_user)
) -> dict:
    """관리자 권한 필요"""

    if current_user.get("role") != "admin":
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="관리자 권한이 필요합니다"
        )

    return current_user
```

```python
"""Authentication API routes"""

from fastapi import APIRouter, Depends, HTTPException, status, Request
from ..auth.models import UserCreate, UserLogin, LoginResponse, TokenPair
from ..auth.service import AuthService
from ..auth.middleware import get_current_active_user
from ..cache_manager import CacheManager

router = APIRouter(prefix="/api/auth", tags=["Authentication"])


@router.post("/register", response_model=dict, status_code=status.HTTP_201_CREATED)
async def register(
    user_data: UserCreate,
    request: Request
):
    """회원가입"""

    auth_service = AuthService(request.app.state.cache_manager)

    try:
        user = await auth_service.create_user(user_data)
        return {
            "message": "회원가입이 완료되었습니다",
            "user": user.dict()
        }

    except ValueError as e:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail=str(e)
        )


@router.post("/login", response_model=LoginResponse)
async def login(
    credentials: UserLogin,
    request: Request
):
    """로그인"""

    auth_service = AuthService(request.app.state.cache_manager)

    try:
        # IP 주소 추출
        ip_address = request.client.host if request.client else None

        result = await auth_service.authenticate_user(credentials, ip_address)

        return LoginResponse(
            user=result["user"],
            tokens=TokenPair(**result["tokens"])
        )

    except ValueError as e:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail=str(e)
        )


@router.post("/logout")
async def logout(
    request: Request,
    current_user: dict = Depends(get_current_active_user)
):
    """로그아웃"""

    # 세션 ID는 토큰에서 추출하거나 별도로 전달받아야 함
    # 여기서는 간단히 모든 세션을 종료하는 방식으로 구현

    auth_service = AuthService(request.app.state.cache_manager)

    # 사용자의 모든 활성 세션 조회 및 삭제
    user_id = current_user["user_id"]
    session_ids = await request.app.state.cache_manager.redis_client.smembers(
        f"user:sessions:{user_id}"
    )

    for session_id in session_ids:
        await auth_service.logout(session_id)

    return {"message": "로그아웃되었습니다"}


@router.get("/me")
async def get_current_user_info(
    current_user: dict = Depends(get_current_active_user)
):
    """현재 사용자 정보 조회"""

    return {
        "user": current_user
    }
```

```python
    # 기본 필터: 사용자 ID

        group_filter = "|".join(group_ids)

    # 필터 조합 (AND 연산)
    filter_expr = " ".join(filter_parts)

    # 검색 쿼리
    query = f"{filter_expr}=>[KNN {top_k} @vector $vec AS score]"
```

### 3.1 API Key 모델

**`src/auth/api_key.py`** (신규 파일)
```python
"""API Key models and service"""

from pydantic import BaseModel, Field
from typing import Optional, List
from datetime import datetime, timedelta
import secrets
import hashlib
import uuid


class APIKeyCreate(BaseModel):
    """API Key 생성 요청"""
    name: str = Field(..., min_length=1, max_length=100)
    permissions: List[str] = Field(default=["read"])  # read, write
    expires_at: Optional[datetime] = None
    rate_limit: int = Field(default=100, ge=1, le=10000)  # 분당 요청 수


class APIKey(BaseModel):
    """API Key 모델"""
    key_id: str
    user_id: str
    key_prefix: str  # 처음 12자만 표시
    name: str
    permissions: List[str]
    created_at: datetime
    expires_at: Optional[datetime] = None
    last_used_at: Optional[datetime] = None
    is_active: bool = True
    rate_limit: int


class APIKeyWithSecret(APIKey):
    """API Key + 실제 키 값 (생성 시에만 반환)"""
    api_key: str  # 전체 키 값


def generate_api_key() -> str:
    """API Key 생성 (chatbot_ + 32자 랜덤 문자열)"""
    random_part = secrets.token_urlsafe(24)  # 32자 base64url
    return f"chatbot_{random_part}"


def hash_api_key(api_key: str) -> str:
    """API Key 해시 (SHA-256)"""
    return hashlib.sha256(api_key.encode()).hexdigest()


class APIKeyService:
    """API Key 관리 서비스"""

    def __init__(self, redis_client):
        self.redis = redis_client

    async def create_api_key(
        self,
        user_id: str,
        key_data: APIKeyCreate
    ) -> APIKeyWithSecret:
        """API Key 생성"""

        # 1. API Key 생성
        api_key = generate_api_key()
        key_hash = hash_api_key(api_key)
        key_id = str(uuid.uuid4())

        # 2. 키 데이터 생성
        now = datetime.utcnow()
        key_obj = APIKeyWithSecret(
            key_id=key_id,
            user_id=user_id,
            api_key=api_key,  # ⚠️ 한 번만 반환됨
            key_prefix=api_key[:12],  # chatbot_abc...
            name=key_data.name,
            permissions=key_data.permissions,
            created_at=now,
            expires_at=key_data.expires_at,
            is_active=True,
            rate_limit=key_data.rate_limit
        )

        # 3. Redis에 저장
        key_dict = {
            "key_id": key_id,
            "user_id": user_id,
            "key_prefix": api_key[:12],
            "name": key_data.name,
            "permissions": json.dumps(key_data.permissions),
            "created_at": now.isoformat(),
            "expires_at": key_data.expires_at.isoformat() if key_data.expires_at else "",
            "is_active": "True",
            "rate_limit": str(key_data.rate_limit)
        }

        # key_id로 조회할 수 있도록 저장
        await self.redis.hset(f"apikey:{key_id}", mapping=key_dict)

        # key_hash로 빠르게 검증할 수 있도록 인덱스 저장
        await self.redis.hset(f"apikey:hash:{key_hash}", mapping=key_dict)

        # 사용자의 API Key 목록에 추가
        await self.redis.sadd(f"user:apikeys:{user_id}", key_id)

        return key_obj

    async def delete_api_key(self, key_id: str, user_id: str) -> bool:
        """API Key 삭제"""

        # 키 조회
        key_data = await self.redis.hgetall(f"apikey:{key_id}")
        if not key_data or key_data["user_id"] != user_id:
            return False

        # 해시 조회를 위해 key_hash 재생성 필요 (하지만 원본 키가 없으므로)
        # 대신 is_active를 False로 변경하는 방식 사용
        await self.redis.hset(f"apikey:{key_id}", "is_active", "False")

        return True

    async def verify_api_key(self, api_key: str) -> Optional[dict]:
        """API Key 검증"""

        key_hash = hash_api_key(api_key)
        key_data = await self.redis.hgetall(f"apikey:hash:{key_hash}")

        if not key_data or key_data.get("is_active") != "True":
            return None

        # 만료일 확인
        if key_data.get("expires_at"):
            expires_at = datetime.fromisoformat(key_data["expires_at"])
            if datetime.utcnow() > expires_at:
                return None

        return key_data

    async def check_rate_limit(self, key_id: str, rate_limit: int) -> bool:
        """Rate Limiting 확인"""

        now = datetime.utcnow()
        minute_key = now.strftime("%Y%m%d%H%M")
        rate_key = f"ratelimit:{key_id}:{minute_key}"

        # 현재 요청 수 확인
        current_count = await self.redis.incr(rate_key)

        # 첫 요청이면 TTL 설정 (2분)
        if current_count == 1:
            await self.redis.expire(rate_key, 120)

        # Rate limit 초과 확인
        return current_count <= rate_limit
```

### 3.2 API Key 인증 미들웨어

**`src/auth/middleware.py`에 추가**
```python
async def authenticate_request(
    request: Request,
    credentials: Optional[HTTPAuthorizationCredentials] = Depends(security)
) -> dict:
    """JWT 또는 API Key 인증"""

    # 1. JWT 토큰 확인
    if credentials:
        user = await get_current_user(request, credentials)
        return {"user_id": user["user_id"], "auth_type": "jwt"}

    # 2. API Key 확인
    api_key = request.headers.get("X-API-Key")
    if api_key:
        api_key_service = APIKeyService(request.app.state.cache_manager.redis_client)

        # API Key 검증
        key_data = await api_key_service.verify_api_key(api_key)
        if not key_data:
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail="유효하지 않은 API Key입니다"
            )

        # Rate Limiting 확인
        rate_ok = await api_key_service.check_rate_limit(
            key_data["key_id"],
            int(key_data["rate_limit"])
        )

        if not rate_ok:
            raise HTTPException(
                status_code=status.HTTP_429_TOO_MANY_REQUESTS,
                detail="요청 한도를 초과했습니다"
            )

        # 권한 확인 (POST, PUT, DELETE는 write 권한 필요)
        permissions = json.loads(key_data["permissions"])
        if request.method in ["POST", "PUT", "DELETE"] and "write" not in permissions:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail="쓰기 권한이 필요합니다"
            )

        return {
            "user_id": key_data["user_id"],
            "key_id": key_data["key_id"],
            "auth_type": "api_key"
        }

    # 3. 인증 정보 없음
    raise HTTPException(
        status_code=status.HTTP_401_UNAUTHORIZED,
        detail="인증이 필요합니다 (JWT 토큰 또는 API Key)"
    )
```

```python
"""Advanced filtering logic"""

from typing import List, Optional
from datetime import datetime
from .models import AdvancedFilter


async def apply_advanced_filter(
    redis_client,
    user_id: str,
    advanced_filter: AdvancedFilter
) -> List[str]:
    """고급 필터 적용하여 문서 목록 반환"""

    # 1. 모든 사용자 문서 조회
    pattern = f"user:{user_id}:document:*"
    eligible_documents = []

    async for key in redis_client.scan_iter(match=pattern):
        metadata = await redis_client.hgetall(key)
        filename = key.split(":")[-1]

        # 2. 날짜 필터 적용
        if advanced_filter.uploaded_after or advanced_filter.uploaded_before:
            uploaded_at = datetime.fromisoformat(metadata["uploaded_at"])

            if advanced_filter.uploaded_after and uploaded_at < advanced_filter.uploaded_after:
                continue
            if advanced_filter.uploaded_before and uploaded_at > advanced_filter.uploaded_before:
                continue

        # 3. 파일 형식 필터
        if advanced_filter.file_types:
            file_type = metadata["file_type"]
            if file_type not in advanced_filter.file_types:
                continue

        # 4. 파일 크기 필터
        if advanced_filter.min_size or advanced_filter.max_size:
            file_size = int(metadata["file_size"])

            if advanced_filter.min_size and file_size < advanced_filter.min_size:
                continue
            if advanced_filter.max_size and file_size > advanced_filter.max_size:
                continue

        # 5. 모든 필터 통과
        eligible_documents.append(filename)

    return eligible_documents
```
